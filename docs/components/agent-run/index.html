Installation

Copy this component into your configured source directory:

$ npx agentcn-ui@latest add agent-run

Or import it from the package:

import { AgentRun } from "@agentcn/react";

Import @agentcn/react/styles.css once in your root layout when using the package. For source installs, import the CSS path from your agentcn.json.

API reference

Props are generated from the component’s TypeScript interface. Surface components also accept native section attributes such as className, style, and aria-label.

PropTypeRequired
titlestringYes
statusRunStatusYes
stepsRunStep[]Yes
onPause() => void | Promise<void>No
onResume() => void | Promise<void>No
onCancel() => void | Promise<void>No

Behavior & integration

Interactive values are controlled by your application. Update the relevant props after a callback succeeds. Async action callbacks disable their controls while pending and display a thrown error without reporting success.

Inspect the component
"use client";

import { Surface, Status, cx, useAction } from "./_shared";
import type { RunStatus, SurfaceProps } from "./_shared";

export interface RunStep {
  id: string;
  label: string;
  status: RunStatus;
  detail?: string;
  duration?: string;
}
export interface AgentRunProps extends SurfaceProps {
  title: string;
  status: RunStatus;
  steps: RunStep[];
  onPause?: () => void | Promise<void>;
  onResume?: () => void | Promise<void>;
  onCancel?: () => void | Promise<void>;
}
export function AgentRun({
  title,
  status,
  steps,
  onPause,
  onResume,
  onCancel,
  ...props
}: AgentRunProps) {
  const { pending, error, run } = useAction();
  const terminal = ["completed", "failed", "cancelled"].includes(status);
  return (
    <Surface
      {...props}
      title={title}
      eyebrow="Agent run"
      actions={<Status value={status} />}
    >
      <ol className="ac-timeline">
        {steps.map((step, i) => (
          <li
            key={step.id}
            className={cx(
              "ac-step",
              step.status === "running" && "ac-step-active",
            )}
          >
            <span className="ac-step-index" aria-hidden="true">
              {step.status === "completed" ? "✓" : i + 1}
            </span>
            <div className="ac-grow">
              <div className="ac-row">
                <span className="ac-medium">{step.label}</span>
                <span className="ac-muted ac-mono">{step.duration}</span>
              </div>
              {step.detail && <p className="ac-muted">{step.detail}</p>}
              <span className="ac-sr-only">{step.status}</span>
            </div>
          </li>
        ))}
      </ol>
      {!terminal && (
        <footer className="ac-footer">
          {status === "running" && onPause && (
            <button
              className="ac-button"
              disabled={pending}
              onClick={() => void run(onPause)}
            >
              Pause run
            </button>
          )}
          {status === "paused" && onResume && (
            <button
              className="ac-button ac-primary"
              disabled={pending}
              onClick={() => void run(onResume)}
            >
              Resume run
            </button>
          )}
          {onCancel && (
            <button
              className="ac-button ac-ghost"
              disabled={pending}
              onClick={() => void run(onCancel)}
            >
              Cancel run
            </button>
          )}
          {pending && <span role="status">Updating…</span>}
        </footer>
      )}
      {error && (
        <p className="ac-error" role="alert">
          {error}
        </p>
      )}
    </Surface>
  );
}
