Installation

Copy this component into your configured source directory:

$ npx agentcn-ui@latest add checkpoint

Or import it from the package:

import { Checkpoint } from "@alvarovillalbaa/agentcn-ui";

Import @alvarovillalbaa/agentcn-ui/styles.css once in your root layout when using the package. For source installs, import the CSS path from your agentcn.json.

API reference

Props are generated from the component’s TypeScript interface. Surface components also accept native section attributes such as className, style, and aria-label.

Behavior & integration

Interactive values are controlled by your application. Update the relevant props after a callback succeeds. Async action callbacks disable their controls while pending and display a thrown error without reporting success.

Keep the server authoritative.

Reviewing a decision in the interface does not authorize an operation. Validate the user, request, current state, and scope on your server before executing a callback’s action.

"use client";

import { Surface, useAction } from "./_shared";
import type { SurfaceProps } from "./_shared";
export interface CheckpointFile {
  path: string;
  additions: number;
  deletions: number;
}
export interface CheckpointProps extends SurfaceProps {
  title: string;
  createdAt: string;
  files: CheckpointFile[];
  onRestore?: () => void | Promise<void>;
  restored?: boolean;
}
export function Checkpoint({
  title,
  createdAt,
  files,
  onRestore,
  restored,
  ...props
}: CheckpointProps) {
  const action = useAction();
  return (
    <Surface
      {...props}
      title={title}
      eyebrow="Run checkpoint"
      actions={<span className="ac-muted ac-small">{createdAt}</span>}
    >
      <ul className="ac-list">
        {files.length === 0 && (
          <li className="ac-empty">No file changes in this checkpoint.</li>
        )}
        {files.map((f) => (
          <li key={f.path} className="ac-row ac-file">
            <code>{f.path}</code>
            <span className="ac-mono">
              <span className="ac-add">+{f.additions}</span>{" "}
              <span className="ac-remove">−{f.deletions}</span>
            </span>
          </li>
        ))}
      </ul>
      {onRestore && (
        <footer className="ac-footer">
          <button
            className="ac-button"
            disabled={action.pending || restored}
            onClick={() => void action.run(onRestore)}
          >
            {restored
              ? "Checkpoint restored"
              : action.pending
                ? "Restoring…"
                : "Restore checkpoint"}
          </button>
          <span className="ac-muted ac-small">
            Review changes before restoring.
          </span>
        </footer>
      )}
      {action.error && (
        <p className="ac-error" role="alert">
          {action.error}
        </p>
      )}
    </Surface>
  );
}
