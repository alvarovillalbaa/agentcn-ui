Installation

Copy this component into your configured source directory:

$ npx agentcn-ui@latest add artifact-panel

Or import it from the package:

import { ArtifactPanel } from "@alvarovillalbaa/agentcn-ui";

Import @alvarovillalbaa/agentcn-ui/styles.css once in your root layout when using the package. For source installs, import the CSS path from your agentcn.json.

API reference

Props are generated from the component’s TypeScript interface. Surface components also accept native section attributes such as className, style, and aria-label.

Behavior & integration

Interactive values are controlled by your application. Update the relevant props after a callback succeeds. Async action callbacks disable their controls while pending and display a thrown error without reporting success.

Artifact content is displayed as escaped text. HTML, JavaScript, and model-generated instructions are never executed by this component.

"use client";

import { Surface, useAction } from "./_shared";
import type { SurfaceProps } from "./_shared";
export interface ArtifactPanelProps extends SurfaceProps {
  name: string;
  language?: string;
  content: string;
  version?: string;
  status?: "streaming" | "ready" | "error";
  error?: string;
  onDownload?: () => void | Promise<void>;
}
/** Text-only by design: model-generated HTML and code are never executed. */
export function ArtifactPanel({
  name,
  language = "text",
  content,
  version,
  status = "ready",
  error,
  onDownload,
  ...props
}: ArtifactPanelProps) {
  const action = useAction();
  return (
    <Surface
      {...props}
      title={name}
      eyebrow="Generated artifact"
      actions={<span className="ac-tag">{version ?? language}</span>}
    >
      <pre
        className="ac-code ac-artifact"
        aria-label={`${name} content`}
        aria-busy={status === "streaming"}
      >
        <code>{content || "Waiting for content…"}</code>
      </pre>
      <footer className="ac-footer">
        <span className="ac-muted" role="status">
          {status === "streaming"
            ? "Generating…"
            : status === "error"
              ? "Generation failed"
              : "Ready to use"}
        </span>
        {onDownload && (
          <button
            className="ac-button ac-small"
            disabled={action.pending || status !== "ready"}
            onClick={() => void action.run(onDownload)}
          >
            Download
          </button>
        )}
      </footer>
      {(error || action.error) && (
        <p className="ac-error" role="alert">
          {error ?? action.error}
        </p>
      )}
    </Surface>
  );
}
