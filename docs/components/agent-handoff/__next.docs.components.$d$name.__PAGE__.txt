  context: string[];
  status: "proposed" | "accepted" | "declined";
  onAccept?: () => void | Promise<void>;
  onDecline?: () => void | Promise<void>;
}
export function AgentHandoff({
  from,
  to,
  reason,
  context,
  status,
  onAccept,
  onDecline,
  ...props
}: AgentHandoffProps) {
  const action = useAction();

    <Surface
      {...props}

    >
      <div className="ac-handoff">

      </details>
      {status === "proposed" && (
        <footer className="ac-footer">
