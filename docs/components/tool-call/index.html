Installation

Copy this component into your configured source directory:

$ npx agentcn-ui@latest add tool-call

Or import it from the package:

import { ToolCall } from "@alvarovillalbaa/agentcn-ui";

Import @alvarovillalbaa/agentcn-ui/styles.css once in your root layout when using the package. For source installs, import the CSS path from your agentcn.json.

API reference

Props are generated from the component’s TypeScript interface. Surface components also accept native section attributes such as className, style, and aria-label.

PropTypeRequired
namestringYes
statusRunStatusYes
inputunknownYes

errorstringNo
durationstringNo
onRetry() => void | Promise<void>No

Behavior & integration

Interactive values are controlled by your application. Update the relevant props after a callback succeeds. Async action callbacks disable their controls while pending and display a thrown error without reporting success.

Inspect the component
"use client";

import { JsonView, Status, Surface, useAction } from "./_shared";
import type { RunStatus, SurfaceProps } from "./_shared";
export interface ToolCallProps extends SurfaceProps {
  name: string;
  status: RunStatus;
  input: unknown;
  output?: unknown;
  error?: string;
  duration?: string;
  onRetry?: () => void | Promise<void>;
}
export function ToolCall({
  name,
  status,
  input,
  output,
  error,
  duration,
  onRetry,
  ...props
}: ToolCallProps) {
  const action = useAction();
  return (
    <Surface
      {...props}
      title={name}
      eyebrow="Tool invocation"
      actions={<Status value={status} />}
    >
      <details className="ac-disclosure" open>
        <summary>
          Input <span className="ac-muted ac-mono">{duration}</span>
        </summary>
        <JsonView value={input} label="Tool input" />
      </details>
      {output !== undefined && (
        <details className="ac-disclosure">
          <summary>Output</summary>
          <JsonView value={output} label="Tool output" />
        </details>
      )}
      {error && (
        <p className="ac-error" role="alert">
          {error}
        </p>
      )}
      {status === "failed" && onRetry && (
        <footer className="ac-footer">
          <button
            className="ac-button"
            disabled={action.pending}
            onClick={() => void action.run(onRetry)}
          >
            {action.pending ? "Retrying…" : "Retry tool"}
          </button>
        </footer>
      )}
      {action.error && (
        <p role="alert" className="ac-error">
          {action.error}
        </p>
      )}
    </Surface>
  );
}
