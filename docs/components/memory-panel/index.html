Installation

Copy this component into your configured source directory:

$ npx agentcn-ui@latest add memory-panel

Or import it from the package:

import { MemoryPanel } from "@agentcn/react";

Import @agentcn/react/styles.css once in your root layout when using the package. For source installs, import the CSS path from your agentcn.json.

API reference

Props are generated from the component’s TypeScript interface. Surface components also accept native section attributes such as className, style, and aria-label.

PropTypeRequired
entriesMemoryEntry[]Yes
onForget(id: string) => void | Promise<void>No

Behavior & integration

Interactive values are controlled by your application. Update the relevant props after a callback succeeds. Async action callbacks disable their controls while pending and display a thrown error without reporting success.

Keep the server authoritative.

Reviewing a decision in the interface does not authorize an operation. Validate the user, request, current state, and scope on your server before executing a callback’s action.

"use client";

import { Surface, useAction } from "./_shared";
import type { SurfaceProps } from "./_shared";
export interface MemoryEntry {
  id: string;
  content: string;
  source: string;
  updatedAt?: string;
  kind: "preference" | "fact" | "instruction";
}
export interface MemoryPanelProps extends SurfaceProps {
  entries: MemoryEntry[];
  onForget?: (id: string) => void | Promise<void>;
}
export function MemoryPanel({ entries, onForget, ...props }: MemoryPanelProps) {
  const action = useAction();
  return (
    <Surface
      {...props}
      title="Agent memory"
      eyebrow="Persistent context"
      actions={
        <span className="ac-muted ac-mono">{entries.length} memories</span>
      }
    >
      {entries.length === 0 ? (
        <p className="ac-empty">
          No saved memories. New memories will appear here.
        </p>
      ) : (
        <ul className="ac-list">
          {entries.map((e) => (
            <li className="ac-memory" key={e.id}>
              <div className="ac-row">
                <span className="ac-tag">{e.kind}</span>
                {onForget && (
                  <button
                    className="ac-button ac-ghost ac-small"
                    disabled={action.pending}
                    aria-label={`Forget memory: ${e.content}`}
                    onClick={() => void action.run(() => onForget(e.id))}
                  >
                    Forget
                  </button>
                )}
              </div>
              <p>{e.content}</p>
              <span className="ac-muted ac-small">
                {e.source}
                {e.updatedAt && ` · ${e.updatedAt}`}
              </span>
            </li>
          ))}
        </ul>
      )}
      {action.error && (
        <p className="ac-error" role="alert">
          {action.error}
        </p>
      )}
    </Surface>
  );
}
